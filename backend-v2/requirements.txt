# =============================================================================
# DEDAN Health — backend runtime dependencies
#
# This file lists ONLY what `main_clinical:app` actually imports at runtime.
# It was verified by importing the application in a clean virtualenv; see
# docs/development.md for the verification command.
#
# Optional/lazy capability modules (billing, hub, chronic care, EMR, the
# LangChain/CrewAI agent graph) are NOT imported by the Clinical API and are
# therefore not required here. See docs/architecture.md.
#
#   python -m venv .venv
#   .venv/bin/pip install -r backend-v2/requirements.txt
# =============================================================================

# --- Web framework ----------------------------------------------------------
fastapi>=0.110
uvicorn[standard]>=0.27
python-multipart>=0.0.9        # multipart/form-data for image uploads
itsdangerous>=2.1              # SessionMiddleware; pulled in by starlette
                                # only under the `full` extra, so it is not a
                                # transitive dependency of uvicorn/fastapi.
                                # main_clinical.py imports SessionMiddleware
                                # unconditionally, so this is required.

# --- Settings & validation --------------------------------------------------
pydantic>=2.6
pydantic-settings>=2.2         # env-driven Settings (app/core/config.py)

# --- Image pipeline ---------------------------------------------------------
Pillow>=10.2                    # image_service.py: decode + quality assessment
numpy>=1.26                     # providers/image_validation.py: brightness,
                                # contrast and blur metrics. Imported at module
                                # load via providers/__init__.py, so it is
                                # required even though it is not obvious from
                                # the route handlers.

# --- AI provider adapters ---------------------------------------------------
# Imported by providers/gemini_provider.py and providers/openai_provider.py.
# Both are only *called* in live mode; importing them in offline mode is safe.
# `google-generativeai` (the legacy SDK) has no Live API at all: its `Client`
# exposes no `.live` attribute, so a realtime session cannot be opened with it.
# Native duplex requires `google-genai`, whose `live.AsyncLive` provides the
# bidirectional session.
google-generativeai>=0.8      # legacy text/image path
google-genai>=1.0             # CURRENT SDK — required for realtime live duplex
openai>=1.30

# --- Persistence ------------------------------------------------------------
# SQLAlchemy 2.0 declarative: the platform models, the database session scope,
# and the notification queue all import it at module load, and
# `main_clinical.create_app()` calls `dedan_platform.db.create_all()` during
# startup -- so this is required by the deployed app, not an optional extra.
#
# This was previously listed only in the "not required" comment block below,
# which was stale: it left a clean `pip install -r requirements.txt` unable to
# import `main_clinical`.
sqlalchemy>=2.0

# --- Test suite -------------------------------------------------------------
httpx>=0.27                     # fastapi.testclient transport
pytest>=8.0
